# 原创运行库探针策略测试使用真实 Lua VM；标准库只属于此测试目标。
# 应用的 Lua 检查器仍仅编译语法，不链接或运行这些标准库。
set(lua_probe_manifest_path "${PROJECT_SOURCE_DIR}/tools/runtime/LUA_POLICY_TEST_SOURCE_MANIFEST.json")
file(READ "${lua_probe_manifest_path}" lua_probe_manifest)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${lua_probe_manifest_path}")
string(JSON lua_probe_file_count LENGTH "${lua_probe_manifest}" files)
math(EXPR lua_probe_last_file "${lua_probe_file_count} - 1")
set(lua_probe_library_sources)
foreach(index RANGE ${lua_probe_last_file})
    string(JSON source GET "${lua_probe_manifest}" files ${index} path)
    string(JSON expected GET "${lua_probe_manifest}" files ${index} sha256)
    if(NOT EXISTS "${AUD_LUA53_SOURCE_DIR}/${source}" OR IS_DIRECTORY "${AUD_LUA53_SOURCE_DIR}/${source}")
        message(FATAL_ERROR "Lua probe policy test dependency missing: ${source}. Prepare the verified Lua 5.3.6 source cache first.")
    endif()
    file(SHA256 "${AUD_LUA53_SOURCE_DIR}/${source}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Lua 5.3.6 probe policy test source SHA256 mismatch: ${source}")
    endif()
    if(source MATCHES "\\.c$")
        list(APPEND lua_probe_library_sources "${AUD_LUA53_SOURCE_DIR}/${source}")
    endif()
endforeach()

    "${PROJECT_SOURCE_DIR}/tools/runtime/RuntimeLuaProbePolicyTests.cpp" ${lua_probe_library_sources})
