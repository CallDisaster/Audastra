触发器 天赋 计算 Talent_Count:
    actions:
        CommentString("先设 AUT_P 为玩家编号；可选数 = 已开放、尚未选择的层数")
        SetVariable(udg_AUT_Open[udg_AUT_P], 0)
        SetVariable(udg_AUT_Free[udg_AUT_P], 0)
        IfThenElseMultiple() {
            if:
                OperatorCompareInteger(udg_AUT_Group[udg_AUT_P], OperatorGreater, 0)
            then:
                ForLoopVarMultiple(udg_AUT_I, 1, 5) {
                    actions:
                        IfThenElseMultiple() {
                            if:
                                OperatorCompareInteger(GetHeroLevel(udg_AUT_Hero[udg_AUT_P]), OperatorGreaterEq, udg_AUT_TierLevel[udg_AUT_I])
                            then:
                                SetVariable(udg_AUT_Open[udg_AUT_P], OperatorInt(udg_AUT_Open[udg_AUT_P], OperatorAdd, 1))
                        }
                        IfThenElseMultiple() {
                            if:
                                OperatorCompareInteger(GetHeroLevel(udg_AUT_Hero[udg_AUT_P]), OperatorGreaterEq, udg_AUT_TierLevel[udg_AUT_I])
                                OperatorCompareInteger(udg_AUT_Picked[OperatorInt(OperatorInt(udg_AUT_P, OperatorMultiply, 8), OperatorAdd, udg_AUT_I)], OperatorEqual, 0)
                            then:
                                SetVariable(udg_AUT_Free[udg_AUT_P], OperatorInt(udg_AUT_Free[udg_AUT_P], OperatorAdd, 1))
                        }
                }
        }

触发器 天赋 绑定 Talent_Bind:
    actions:
        CommentString("先设 AUT_Unit 为英雄；类型登记过时成为其所属玩家的天赋英雄（换英雄时已选结果保留）")
        SetVariable(udg_AUT_Found, 0)
        ForLoopVarMultiple(udg_AUT_I, 1, udg_AUT_HeroCount) {
            actions:
                IfThenElseMultiple() {
                    if:
                        OperatorCompareUnitCode(GetUnitTypeId(udg_AUT_Unit), OperatorEqualENE, udg_AUT_HeroType[udg_AUT_I])
                    then:
                        SetVariable(udg_AUT_Found, udg_AUT_I)
                }
        }
        IfThenElseMultiple() {
            if:
                OperatorCompareInteger(udg_AUT_Found, OperatorGreater, 0)
            then:
                SetVariable(udg_AUT_P, GetConvertedPlayerId(GetOwningPlayer(udg_AUT_Unit)))
                SetVariable(udg_AUT_Hero[udg_AUT_P], udg_AUT_Unit)
                SetVariable(udg_AUT_Group[udg_AUT_P], udg_AUT_Found)
                ConditionalTriggerExecute(`天赋 计算 Talent_Count`)
                ConditionalTriggerExecute(`天赋 刷新 Talent_Refresh`)
        }
触发器 天赋 候选 Talent_Candidate:
    actions:
        IfThenElseMultiple() {
            if:
                OperatorCompareBoolean(IsUnitType(udg_AUT_Unit, UnitTypeHero), OperatorEqualENE, true)
                OperatorCompareBoolean(IsUnitIllusionBJ(udg_AUT_Unit), OperatorEqualENE, false)
                OperatorComparePlayerControl(GetPlayerController(GetOwningPlayer(udg_AUT_Unit)), OperatorEqualENE, MapControlUser)
                OperatorComparePlayerSlotStatus(GetPlayerSlotState(GetOwningPlayer(udg_AUT_Unit)), OperatorEqualENE, PlayerSlotStatePlaying)
                OperatorCompareInteger(udg_AUT_Group[GetConvertedPlayerId(GetOwningPlayer(udg_AUT_Unit))], OperatorEqual, 0)
            then:
                ConditionalTriggerExecute(`天赋 绑定 Talent_Bind`)
        }
触发器 天赋 初始化 Talent_Init (init):
    actions:
        ConditionalTriggerExecute(`天赋 登记 Talent_Register`)
        SetVariable(udg_AUT_TierLevel[1], 5)
        SetVariable(udg_AUT_TierLevel[2], 10)
        SetVariable(udg_AUT_TierLevel[3], 15)
        SetVariable(udg_AUT_TierLevel[4], 20)
        SetVariable(udg_AUT_TierLevel[5], 25)
        CommentString("同步单位：每个用户玩家每个选项一个，透明、无敌、暂停、不占人口，放在地图左下角之外")
        SetVariable(udg_AUT_Point, Location(OperatorReal(GetRectMinX(GetEntireMapRect()), OperatorAdd, 64), OperatorReal(GetRectMinY(GetEntireMapRect()), OperatorAdd, 64)))
        ForLoopVarMultiple(udg_AUT_P, 1, 12) {
            actions:
                IfThenElseMultiple() {
                    if:
                        OperatorComparePlayerControl(GetPlayerController(ConvertedPlayer(udg_AUT_P)), OperatorEqualENE, MapControlUser)
                        OperatorComparePlayerSlotStatus(GetPlayerSlotState(ConvertedPlayer(udg_AUT_P)), OperatorEqualENE, PlayerSlotStatePlaying)
                    then:
                        ForLoopVarMultiple(udg_AUT_T, 1, 5) {
                            actions:
                                ForLoopVarMultiple(udg_AUT_O, 1, 3) {
                                    actions:
                                        CreateNUnitsAtLoc(1, 'hfoo', ConvertedPlayer(udg_AUT_P), udg_AUT_Point, 0)
                                        SetVariable(udg_AUT_Sync[OperatorInt(OperatorInt(OperatorInt(OperatorInt(udg_AUT_P, OperatorSubtract, 1), OperatorMultiply, 28), OperatorAdd, OperatorInt(OperatorInt(udg_AUT_T, OperatorSubtract, 1), OperatorMultiply, 4)), OperatorAdd, udg_AUT_O)], GetLastCreatedUnit())
                                        SetUnitUseFoodBJ(EnableDisableDisable, GetLastCreatedUnit())
                                        SetUnitInvulnerable(GetLastCreatedUnit(), InvulnerabilityInvulnerable)
                                        SetUnitPathing(GetLastCreatedUnit(), PathingOff)
                                        SetUnitPositionLoc(GetLastCreatedUnit(), udg_AUT_Point)
                                        SetUnitVertexColorBJ(GetLastCreatedUnit(), 100, 100, 100, 100)
                                        PauseUnitBJ(PauseUnpauseOptionPause, GetLastCreatedUnit())
                                }
                        }
                }
        }
        RemoveLocation(udg_AUT_Point)
        ForGroupMultiple(GetUnitsInRectAll(GetPlayableMapRect())) {
            actions:
                SetVariable(udg_AUT_Unit, GetEnumUnit())
                ConditionalTriggerExecute(`天赋 候选 Talent_Candidate`)
        }
        ConditionalTriggerExecute(`天赋 刷新 Talent_Refresh`)

触发器 天赋 选择 Talent_Pick:
    events:
        TriggerRegisterPlayerSelectionEventBJ(Player00, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player01, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player02, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player03, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player04, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player05, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player06, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player07, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player08, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player09, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player10, SelDeselOptionSel)
        TriggerRegisterPlayerSelectionEventBJ(Player11, SelDeselOptionSel)
    conditions:
        OperatorCompareUnitCode(GetUnitTypeId(GetTriggerUnit()), OperatorEqualENE, 'hfoo')
    actions:
        CommentString("同步单位被选中：所有玩家都会运行，这里写入同步的选择结果")
        SetVariable(udg_AUT_P, GetConvertedPlayerId(GetTriggerPlayer()))
        SetVariable(udg_AUT_T, 0)
        SetVariable(udg_AUT_O, 0)
        ForLoopVarMultiple(udg_AUT_I, 1, 5) {
            actions:
                ForLoopVarMultiple(udg_AUT_J, 1, 3) {
                    actions:
                        IfThenElseMultiple() {
                            if:
                                OperatorCompareUnit(GetTriggerUnit(), OperatorEqualENE, udg_AUT_Sync[OperatorInt(OperatorInt(OperatorInt(OperatorInt(udg_AUT_P, OperatorSubtract, 1), OperatorMultiply, 28), OperatorAdd, OperatorInt(OperatorInt(udg_AUT_I, OperatorSubtract, 1), OperatorMultiply, 4)), OperatorAdd, udg_AUT_J)])
                            then:
                                SetVariable(udg_AUT_T, udg_AUT_I)
                                SetVariable(udg_AUT_O, udg_AUT_J)
                        }
                }
        }
        IfThenElseMultiple() {
            if:
                OperatorCompareInteger(udg_AUT_T, OperatorGreater, 0)
            then:
                SelectUnitRemoveForPlayer(GetTriggerUnit(), GetTriggerPlayer())
                IfThenElseMultiple() {
                    if:
                        OperatorCompareInteger(udg_AUT_Group[udg_AUT_P], OperatorGreater, 0)
                        OperatorCompareInteger(GetHeroLevel(udg_AUT_Hero[udg_AUT_P]), OperatorGreaterEq, udg_AUT_TierLevel[udg_AUT_T])
                        OperatorCompareInteger(udg_AUT_Picked[OperatorInt(OperatorInt(udg_AUT_P, OperatorMultiply, 8), OperatorAdd, udg_AUT_T)], OperatorEqual, 0)
                        OperatorCompareString(udg_AUT_Name[OperatorInt(OperatorInt(OperatorInt(OperatorInt(udg_AUT_Group[udg_AUT_P], OperatorSubtract, 1), OperatorMultiply, 28), OperatorAdd, OperatorInt(OperatorInt(udg_AUT_T, OperatorSubtract, 1), OperatorMultiply, 4)), OperatorAdd, udg_AUT_O)], OperatorNotEqualENE, "")
                    then:
                        SetVariable(udg_AUT_K, OperatorInt(OperatorInt(OperatorInt(OperatorInt(udg_AUT_Group[udg_AUT_P], OperatorSubtract, 1), OperatorMultiply, 28), OperatorAdd, OperatorInt(OperatorInt(udg_AUT_T, OperatorSubtract, 1), OperatorMultiply, 4)), OperatorAdd, udg_AUT_O))
                        SetVariable(udg_AUT_Picked[OperatorInt(OperatorInt(udg_AUT_P, OperatorMultiply, 8), OperatorAdd, udg_AUT_T)], udg_AUT_O)
                        IfThenElseMultiple() {
                            if:
                                OperatorCompareBoolean(udg_AUT_Learns[udg_AUT_K], OperatorEqualENE, true)
                            then:
                                UnitAddAbilityBJ(udg_AUT_Learn[udg_AUT_K], udg_AUT_Hero[udg_AUT_P])
                        }
                        ConditionalTriggerExecute(`天赋 计算 Talent_Count`)
                        SetVariable(udg_AUT_PickUnit, udg_AUT_Hero[udg_AUT_P])
                        SetVariable(udg_AUT_PickTier, udg_AUT_T)
                        SetVariable(udg_AUT_PickOption, udg_AUT_O)
                        ConditionalTriggerExecute(`天赋 刷新 Talent_Refresh`)
                }
        }
