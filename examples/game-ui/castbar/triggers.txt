        SetVariable(udg_AUC_Reg, 0)

触发器 读条 开始 Cast_Begin:
    events:
        TriggerRegisterAnyUnitEventBJ(PlayerUnitEventSpellChannel)
    actions:
        SetVariable(udg_AUC_FindAbility, GetSpellAbilityId())
        ConditionalTriggerExecute(`读条 查找技能 Cast_FindAbility`)
        IfThenElseMultiple() {
            if:
                OperatorCompareInteger(udg_AUC_Reg, OperatorGreater, 0)
            then:
                SetVariable(udg_AUC_FindUnit, GetTriggerUnit())
                ConditionalTriggerExecute(`读条 查找施法者 Cast_FindSlot`)
                IfThenElseMultiple() {
                    if:
                        OperatorCompareInteger(udg_AUC_Slot, OperatorEqual, 0)
                    then:
                        SetVariable(udg_AUC_Count, OperatorInt(udg_AUC_Count, OperatorAdd, 1))
                        SetVariable(udg_AUC_Slot, udg_AUC_Count)
                }
                SetVariable(udg_AUC_Unit[udg_AUC_Slot], GetTriggerUnit())
                SetVariable(udg_AUC_SlotReg[udg_AUC_Slot], udg_AUC_Reg)
                SetVariable(udg_AUC_Ability[udg_AUC_Slot], GetSpellAbilityId())
                SetVariable(udg_AUC_Level[udg_AUC_Slot], GetUnitAbilityLevel(GetTriggerUnit(), GetSpellAbilityId()))
                SetVariable(udg_AUC_Target[udg_AUC_Slot], GetSpellTargetUnit())
                SetVariable(udg_AUC_Point, GetSpellTargetLoc())
                SetVariable(udg_AUC_X[udg_AUC_Slot], GetLocationX(udg_AUC_Point))
                SetVariable(udg_AUC_Y[udg_AUC_Slot], GetLocationY(udg_AUC_Point))
                RemoveLocation(udg_AUC_Point)
                SetVariable(udg_AUC_Ticks[udg_AUC_Slot], 0)
                SetVariable(udg_AUC_Stage[udg_AUC_Slot], 1)
                SetVariable(udg_AUC_Start[udg_AUC_Slot], TimerGetElapsed(udg_AUC_Clock))
                SetVariable(udg_AUC_Length[udg_AUC_Slot], udg_AUC_RegCast[udg_AUC_Reg])
        }
触发器 读条 生效 Cast_Effect:
    events:
        TriggerRegisterAnyUnitEventBJ(PlayerUnitEventSpellEffect)
    actions:
        SetVariable(udg_AUC_FindUnit, GetTriggerUnit())
        ConditionalTriggerExecute(`读条 查找施法者 Cast_FindSlot`)
        IfThenElseMultiple() {
            if:
                OperatorCompareInteger(udg_AUC_Slot, OperatorGreater, 0)
                OperatorCompareAbilityId(udg_AUC_Ability[udg_AUC_Slot], OperatorEqualENE, GetSpellAbilityId())
                OperatorCompareInteger(udg_AUC_Stage[udg_AUC_Slot], OperatorEqual, 1)
            then:
                IfThenElseMultiple() {
                    if:
                        OperatorCompareReal(udg_AUC_RegChannel[udg_AUC_SlotReg[udg_AUC_Slot]], OperatorGreater, 0.00)
                    then:
                        SetVariable(udg_AUC_Stage[udg_AUC_Slot], 2)
                        SetVariable(udg_AUC_Start[udg_AUC_Slot], TimerGetElapsed(udg_AUC_Clock))
                        SetVariable(udg_AUC_Length[udg_AUC_Slot], udg_AUC_RegChannel[udg_AUC_SlotReg[udg_AUC_Slot]])
                    else:
                        SetVariable(udg_AUC_Phase, 1)
                        ConditionalTriggerExecute(`读条 派发 Cast_Dispatch`)
                }
        }
触发器 读条 结束 Cast_Finish:
    events:
        TriggerRegisterAnyUnitEventBJ(PlayerUnitEventSpellFinish)
    actions:
        SetVariable(udg_AUC_FindUnit, GetTriggerUnit())
        ConditionalTriggerExecute(`读条 查找施法者 Cast_FindSlot`)
        IfThenElseMultiple() {
            if:
                OperatorCompareInteger(udg_AUC_Slot, OperatorGreater, 0)
                OperatorCompareAbilityId(udg_AUC_Ability[udg_AUC_Slot], OperatorEqualENE, GetSpellAbilityId())
                OperatorCompareInteger(udg_AUC_Stage[udg_AUC_Slot], OperatorEqual, 2)
            then:
                IfThenElseMultiple() {
                    if:
                        OperatorCompareReal(udg_AUC_RegTick[udg_AUC_SlotReg[udg_AUC_Slot]], OperatorGreater, 0.00)
                        OperatorCompareInteger(udg_AUC_Ticks[udg_AUC_Slot], OperatorLess, R2I(OperatorReal(OperatorReal(udg_AUC_RegChannel[udg_AUC_SlotReg[udg_AUC_Slot]], OperatorDivide, udg_AUC_RegTick[udg_AUC_SlotReg[udg_AUC_Slot]]), OperatorAdd, 0.001)))
                    then:
                        SetVariable(udg_AUC_Ticks[udg_AUC_Slot], OperatorInt(udg_AUC_Ticks[udg_AUC_Slot], OperatorAdd, 1))
                        SetVariable(udg_AUC_Phase, 2)
                        ConditionalTriggerExecute(`读条 派发 Cast_Dispatch`)
                        SetVariable(udg_AUC_FindUnit, GetTriggerUnit())
                        ConditionalTriggerExecute(`读条 查找施法者 Cast_FindSlot`)
                }
                IfThenElseMultiple() {
                    if:
                        OperatorCompareInteger(udg_AUC_Slot, OperatorGreater, 0)
                        OperatorCompareAbilityId(udg_AUC_Ability[udg_AUC_Slot], OperatorEqualENE, GetSpellAbilityId())
                        OperatorCompareInteger(udg_AUC_Stage[udg_AUC_Slot], OperatorEqual, 2)
                    then:
                        SetVariable(udg_AUC_Phase, 1)
                        ConditionalTriggerExecute(`读条 派发 Cast_Dispatch`)
                }
        }

触发器 读条 刷新 Cast_Refresh:
    events:
        TriggerRegisterTimerEventPeriodic(0.03)
    actions:
        SetVariable(udg_AUC_Now, TimerGetElapsed(udg_AUC_Clock))
        ForLoopVarMultiple(udg_AUC_TickI, 1, udg_AUC_Count) {
            actions:
                SetVariable(udg_AUC_Slot, udg_AUC_TickI)
                IfThenElseMultiple() {
                    if:
                        OperatorCompareBoolean(IsUnitDeadBJ(udg_AUC_Unit[udg_AUC_Slot]), OperatorEqualENE, true)
                    then:
                        SetVariable(udg_AUC_Phase, 3)
                        ConditionalTriggerExecute(`读条 派发 Cast_Dispatch`)
                    else:
                        IfThenElseMultiple() {
                            if:
                                OperatorCompareInteger(udg_AUC_Stage[udg_AUC_Slot], OperatorEqual, 2)
                                OperatorCompareReal(udg_AUC_RegTick[udg_AUC_SlotReg[udg_AUC_Slot]], OperatorGreater, 0.00)
                                OperatorCompareInteger(udg_AUC_Ticks[udg_AUC_Slot], OperatorLess, R2I(OperatorReal(OperatorReal(udg_AUC_RegChannel[udg_AUC_SlotReg[udg_AUC_Slot]], OperatorDivide, udg_AUC_RegTick[udg_AUC_SlotReg[udg_AUC_Slot]]), OperatorAdd, 0.001)))
                                OperatorCompareReal(OperatorReal(udg_AUC_Now, OperatorSubtract, udg_AUC_Start[udg_AUC_Slot]), OperatorGreaterEq, OperatorReal(I2R(OperatorInt(udg_AUC_Ticks[udg_AUC_Slot], OperatorAdd, 1)), OperatorMultiply, udg_AUC_RegTick[udg_AUC_SlotReg[udg_AUC_Slot]]))
                            then:
                                SetVariable(udg_AUC_Ticks[udg_AUC_Slot], OperatorInt(udg_AUC_Ticks[udg_AUC_Slot], OperatorAdd, 1))
                                SetVariable(udg_AUC_Phase, 2)
                                ConditionalTriggerExecute(`读条 派发 Cast_Dispatch`)
                        }
                }
        }
触发器 读条 显示 Cast_Show:
    events:
        TriggerRegisterTimerEventPeriodic(0.03)
    actions:
        CommentString("只改本机界面；同步数据只在施法事件里改")
        SetVariable(udg_AUC_ShowNow, TimerGetElapsed(udg_AUC_Clock))
        SetVariable(udg_AUC_ShowSlot, 0)
        ForLoopVarMultiple(udg_AUC_ShowI, 1, udg_AUC_Count) {
            actions:
                IfThenElseMultiple() {
                    if:
                        OperatorComparePlayer(GetOwningPlayer(udg_AUC_Unit[udg_AUC_ShowI]), OperatorEqualENE, GetLocalPlayer())
                        OperatorCompareReal(udg_AUC_Length[udg_AUC_ShowI], OperatorGreater, 0.00)
                    then:
                        SetVariable(udg_AUC_ShowSlot, udg_AUC_ShowI)
                }
        }
        IfThenElseMultiple() {
            if:
                OperatorCompareInteger(udg_AUC_ShowSlot, OperatorGreater, 0)
            then:
                SetVariable(udg_AUC_ShowRatio, OperatorReal(OperatorReal(udg_AUC_ShowNow, OperatorSubtract, udg_AUC_Start[udg_AUC_ShowSlot]), OperatorDivide, udg_AUC_Length[udg_AUC_ShowSlot]))
                IfThenElseMultiple() {
                    if:
                        OperatorCompareReal(udg_AUC_ShowRatio, OperatorGreater, 1.00)
                    then:
                        SetVariable(udg_AUC_ShowRatio, 1.00)
                }
                IfThenElseMultiple() {
                    if:
                        OperatorCompareInteger(udg_AUC_Stage[udg_AUC_ShowSlot], OperatorEqual, 2)
                    then:
                        SetVariable(udg_AUC_ShowRatio, OperatorReal(1.00, OperatorSubtract, udg_AUC_ShowRatio))
                }
                SetVariable(udg_AUIV_castbar_shown, true)
                SetVariable(udg_AUIV_castbar_progress, udg_AUC_ShowRatio)
                SetVariable(udg_AUIV_castbar_text, GetAbilityName(udg_AUC_Ability[udg_AUC_ShowSlot]))
                SetVariable(udg_AUIV_castbar_remaining, OperatorReal(udg_AUC_Length[udg_AUC_ShowSlot], OperatorSubtract, OperatorReal(udg_AUC_ShowNow, OperatorSubtract, udg_AUC_Start[udg_AUC_ShowSlot])))
                IfThenElseMultiple() {
                    if:
                        OperatorCompareReal(udg_AUIV_castbar_remaining, OperatorLess, 0.00)
                    then:
                        SetVariable(udg_AUIV_castbar_remaining, 0.00)
                }
                IfThenElseMultiple() {
                    if:
                        OperatorCompareInteger(udg_AUC_Stage[udg_AUC_ShowSlot], OperatorEqual, 2)
                        OperatorCompareInteger(udg_AUC_Shown, OperatorNotEqual, 2)
                    then:
                        SetVariable(udg_AUC_Shown, 2)
                        ConditionalTriggerExecute(`读条条 状态 引导 castbar_State_channel`)
                }
                IfThenElseMultiple() {
                    if:
                        OperatorCompareInteger(udg_AUC_Stage[udg_AUC_ShowSlot], OperatorEqual, 1)
                        OperatorCompareInteger(udg_AUC_Shown, OperatorNotEqual, 1)
                    then:
                        SetVariable(udg_AUC_Shown, 1)
                        ConditionalTriggerExecute(`读条条 基础状态 castbar_BaseState`)
                }
                ConditionalTriggerExecute(`读条条 更新 castbar_Update`)
            else:
                IfThenElseMultiple() {
                    if:
                        OperatorCompareReal(udg_AUC_FlashUntil[GetConvertedPlayerId(GetLocalPlayer())], OperatorGreater, udg_AUC_ShowNow)
                    then:
                        IfThenElseMultiple() {
                            if:
                                OperatorCompareInteger(udg_AUC_Shown, OperatorNotEqual, 3)
                            then:
                                SetVariable(udg_AUC_Shown, 3)
                                SetVariable(udg_AUIV_castbar_shown, true)
                                SetVariable(udg_AUIV_castbar_progress, 1.00)
                                SetVariable(udg_AUIV_castbar_text, udg_AUC_FlashName[GetConvertedPlayerId(GetLocalPlayer())])
                                SetVariable(udg_AUIV_castbar_remaining, 0.00)
                                ConditionalTriggerExecute(`读条条 状态 打断 castbar_State_interrupt`)
                                ConditionalTriggerExecute(`读条条 更新 castbar_Update`)
                        }
                    else:
                        IfThenElseMultiple() {
                            if:
                                OperatorCompareInteger(udg_AUC_Shown, OperatorNotEqual, 0)
                            then:
                                SetVariable(udg_AUC_Shown, 0)
                                SetVariable(udg_AUIV_castbar_shown, false)
                                ConditionalTriggerExecute(`读条条 基础状态 castbar_BaseState`)
                                ConditionalTriggerExecute(`读条条 更新 castbar_Update`)
                        }
                }
        }
