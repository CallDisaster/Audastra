cmake_minimum_required(VERSION 3.24)
project(AudastraJapi VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(audastra-japi STATIC src/Protocol.cpp src/Host.cpp src/Events.cpp src/Dispatch.cpp)
target_include_directories(audastra-japi PUBLIC include)
target_compile_features(audastra-japi PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(audastra-japi PRIVATE /EHsc /W4 /permissive- /utf-8)
endif()
if(WIN32)
    add_library(audastra-japi-loader STATIC src/WindowsLoader.cpp)
    target_link_libraries(audastra-japi-loader PUBLIC audastra-japi PRIVATE bcrypt advapi32)
    target_compile_definitions(audastra-japi-loader PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(audastra-japi-loader PRIVATE /EHsc /W4 /permissive- /utf-8)
    endif()
endif()
option(AJ_BUILD_TESTS "Build standalone protocol and DLL contract tests" ${PROJECT_IS_TOP_LEVEL})
option(AJ_BUILD_SAMPLE "Build the example module (does not install or authorize it)" ${PROJECT_IS_TOP_LEVEL})
set(AJ_GENERATED_CONTRACT_HEADER "" CACHE FILEPATH "Optional generated .contract.h to compile-check")
if(AJ_BUILD_SAMPLE)
    add_library(audastra-japi-demo SHARED sample/Demo.c)
    target_include_directories(audastra-japi-demo PRIVATE include)
    set_target_properties(audastra-japi-demo PROPERTIES PREFIX "" OUTPUT_NAME "demo")
    if(MSVC)
        target_compile_options(audastra-japi-demo PRIVATE /utf-8 /W4)
    endif()
endif()
if(AJ_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(audastra-japi-tests tests/ProtocolTests.cpp tests/HostTests.cpp)
    target_link_libraries(audastra-japi-tests PRIVATE audastra-japi Threads::Threads)
    if(MSVC)
        target_compile_options(audastra-japi-tests PRIVATE /EHsc /W4 /permissive- /utf-8)
    endif()
    add_test(NAME audastra-japi-contract COMMAND audastra-japi-tests)
    set_tests_properties(audastra-japi-contract PROPERTIES TIMEOUT 30)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME audastra-module-boot-retry COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_module_boot.py")
    set_tests_properties(audastra-module-boot-retry PROPERTIES TIMEOUT 30)
    add_test(NAME audastra-japi-codegen COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_generate.py")
    set_tests_properties(audastra-japi-codegen PROPERTIES TIMEOUT 30)
    if(AJ_GENERATED_CONTRACT_HEADER)
        add_executable(audastra-japi-generated-tests tests/GeneratedContractTests.cpp)
        target_link_libraries(audastra-japi-generated-tests PRIVATE audastra-japi)
        target_compile_definitions(audastra-japi-generated-tests PRIVATE "AJ_GENERATED_CONTRACT_HEADER=\"${AJ_GENERATED_CONTRACT_HEADER}\"")
        if(MSVC)
            target_compile_options(audastra-japi-generated-tests PRIVATE /EHsc /utf-8 /W4)
        endif()
        add_test(NAME audastra-japi-generated-contract COMMAND audastra-japi-generated-tests)
        set_tests_properties(audastra-japi-generated-contract PROPERTIES TIMEOUT 10)
    endif()
    if(TARGET War3MapEditor::Jass)
        # 执行真实 UI/JAPI 源函数的受限控制流；底层请求由受控替身提供，不冒充游戏 VM。
        add_test(NAME audastra-ui-connect-retry COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ui_connect.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/ui-library/AudastraUI/jass/AudastraUI.j"
            "${CMAKE_CURRENT_SOURCE_DIR}/jass/AudastraJapi.j")
        set_tests_properties(audastra-ui-connect-retry PROPERTIES TIMEOUT 30)
        # 手动专用实机样例打包器；不自动启动游戏，绝不解析黑盒 AI。
        add_executable(audastra-japi-map-fixture EXCLUDE_FROM_ALL tests/MapRuntimeFixture.cpp)
        target_link_libraries(audastra-japi-map-fixture PRIVATE War3MapEditor::Jass War3MapEditor::Core)
        target_compile_options(audastra-japi-map-fixture PRIVATE /EHsc /utf-8 /W4)
        target_compile_definitions(audastra-japi-map-fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        set_property(TARGET audastra-japi-map-fixture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
        add_executable(audastra-japi-jass-tests tests/JassSourceTests.cpp)
        target_link_libraries(audastra-japi-jass-tests PRIVATE War3MapEditor::Jass War3MapEditor::ScriptBuild)
        target_compile_options(audastra-japi-jass-tests PRIVATE /EHsc /utf-8 /W4)
        # The editor libraries use its own CRT setting; don't mix the standalone SDK /MT policy here.
        set_property(TARGET audastra-japi-jass-tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
        add_test(NAME audastra-japi-jass COMMAND audastra-japi-jass-tests "${CMAKE_CURRENT_SOURCE_DIR}/jass/AudastraJapi.j")
        set_tests_properties(audastra-japi-jass PROPERTIES TIMEOUT 30)
        add_test(NAME audastra-japi-smoke-jass COMMAND audastra-japi-jass-tests
            "${CMAKE_CURRENT_SOURCE_DIR}/jass/AudastraJapi.j" "${CMAKE_CURRENT_SOURCE_DIR}/sample/StringBridgeSmoke.j")
        set_tests_properties(audastra-japi-smoke-jass PROPERTIES TIMEOUT 30)
    endif()
    if(WIN32 AND AJ_BUILD_SAMPLE)
        add_library(audastra-japi-worker-fixture SHARED tests/WorkerFixture.cpp)
        target_include_directories(audastra-japi-worker-fixture PRIVATE include)
        target_compile_definitions(audastra-japi-worker-fixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        set_target_properties(audastra-japi-worker-fixture PROPERTIES PREFIX "" OUTPUT_NAME "worker")
        if(MSVC)
            target_compile_options(audastra-japi-worker-fixture PRIVATE /EHsc /utf-8 /W4)
        endif()
        add_executable(audastra-japi-loader-tests tests/LoaderTests.cpp)
        target_link_libraries(audastra-japi-loader-tests PRIVATE audastra-japi-loader bcrypt)
        target_compile_definitions(audastra-japi-loader-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        if(MSVC)
            target_compile_options(audastra-japi-loader-tests PRIVATE /EHsc /utf-8 /W4)
        endif()
        add_dependencies(audastra-japi-loader-tests audastra-japi-demo audastra-japi-worker-fixture)
        add_test(NAME audastra-japi-loader COMMAND audastra-japi-loader-tests $<TARGET_FILE:audastra-japi-demo> $<TARGET_FILE:audastra-japi-worker-fixture>)
        set_tests_properties(audastra-japi-loader PROPERTIES TIMEOUT 30)
    endif()
endif()
