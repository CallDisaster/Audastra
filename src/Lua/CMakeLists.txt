set(AUD_LUA53_SOURCE_DIR "" CACHE PATH "Verified official Lua 5.3.6 src cache directory")
file(READ "${PROJECT_SOURCE_DIR}/third_party/lua53/SOURCE_MANIFEST.json" lua_manifest)
string(JSON lua_file_count LENGTH "${lua_manifest}" files)
math(EXPR lua_last_file "${lua_file_count} - 1")
foreach(index RANGE ${lua_last_file})
    string(JSON source GET "${lua_manifest}" files ${index} path)
    string(JSON expected GET "${lua_manifest}" files ${index} sha256)
    if(NOT EXISTS "${AUD_LUA53_SOURCE_DIR}/${source}" OR IS_DIRECTORY "${AUD_LUA53_SOURCE_DIR}/${source}")
        message(FATAL_ERROR "Lua syntax dependency missing. Run tools/dev/prepare-update-deps.ps1 or prepare-lua-check-deps.ps1 and pass AUD_LUA53_SOURCE_DIR.")
    endif()
    file(SHA256 "${AUD_LUA53_SOURCE_DIR}/${source}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Lua 5.3.6 source SHA256 mismatch: ${source}")
    endif()
endforeach()
set(lua_core_sources)
foreach(source IN ITEMS lapi lcode lctype ldebug ldo ldump lfunc lgc llex lmem lobject
    lopcodes lparser lstate lstring ltable ltm lundump lvm lzio)
    list(APPEND lua_core_sources "${AUD_LUA53_SOURCE_DIR}/${source}.c")
endforeach()
add_library(audastra-lua53-core STATIC ${lua_core_sources})
target_include_directories(audastra-lua53-core PUBLIC "${AUD_LUA53_SOURCE_DIR}")
target_compile_definitions(audastra-lua53-core PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(War3MapLua STATIC LuaSyntax.cpp LuaSemantic.cpp)
add_library(War3MapEditor::Lua ALIAS War3MapLua)
target_include_directories(War3MapLua PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(War3MapLua PRIVATE audastra-lua53-core)
target_compile_options(War3MapLua PRIVATE /W4 /permissive- /utf-8)
