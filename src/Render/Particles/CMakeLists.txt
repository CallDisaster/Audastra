cmake_minimum_required(VERSION 3.24)
get_filename_component(_particle_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ForgeParticlePreview LANGUAGES CXX)
    include("${_particle_root}/cmake/War3RenderCore.cmake")
endif()
add_library(forge-particle-preview STATIC ParticleRead.cpp ParticleTrack.cpp ParticleSimulation.cpp ParticleGeometry.cpp ParticleInput.cpp)
target_compile_features(forge-particle-preview PUBLIC cxx_std_20)
target_include_directories(forge-particle-preview PUBLIC "${_particle_root}/include" "${_particle_root}/src/Render")
if(MSVC)
    target_compile_options(forge-particle-preview PRIVATE /W4 /WX /permissive- /utf-8 /fp:precise)
else()
    target_compile_options(forge-particle-preview PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    target_sources(forge-particle-preview PRIVATE "${_particle_root}/src/Core/Types.cpp")
    enable_testing()
    include(TestTargets.cmake)
elseif(TARGET War3MapEditor::Core)
    target_link_libraries(forge-particle-preview PUBLIC War3MapEditor::Core)
else()
    # 独立 Animation/Win32 检查复用已登记的 Types，不拉入整个 Core 或 Qt。
    target_link_libraries(forge-particle-preview PUBLIC forge-animation-reference)
endif()

target_link_libraries(forge-particle-preview PUBLIC War3RenderCore::Contract)
